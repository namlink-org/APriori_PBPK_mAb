Purpose

This report documents the inputs used for the reduced sparse monkey PK study design and displays the previously completed fold-error analysis. It does not rerun the PK model or the study-design simulations.

The analysis is predefined for atezolizumab, avelumab, nivolumab, and pembrolizumab. Run 4 top-down estimates and relative standard errors (RSEs) are read directly from the final monkey-model fit. Parameter uncertainty is represented by those RSEs, and 30% interindividual variability (IIV) is specified independently for CL, V1, Q, and V2.

Run 4 parameter inputs

run4_fits <- readRDS(run4_file)

missing_molecules <- setdiff(MOLECULES, names(run4_fits))
if (length(missing_molecules) > 0L) {
  stop(
    "Final Run 4 fits are missing for: ",
    paste(missing_molecules, collapse = ", ")
  )
}

parameter_map <- tribble(
  ~Parameter, ~Theta_name, ~Log_parameter,
  "CL",       "cl",        "lcl",
  "V1",       "vc",        "lvc",
  "Q",        "q",         "lq",
  "V2",       "vp",        "lvp"
)

extract_run4_parameters <- function(molecule) {
  fit <- run4_fits[[molecule]]$fit
  estimated_parameters <- names(fit$opt$par)
  rse <- fit$rse_percent

  parameter_map |>
    mutate(
      Molecule = molecule,
      Estimate = map_dbl(Theta_name, ~ unname(fit$theta[[.x]])),
      RSE_percent = map_dbl(
        Log_parameter,
        ~ if (.x %in% names(rse)) unname(rse[[.x]]) else NA_real_
      ),
      Run4_status = if_else(
        Log_parameter %in% estimated_parameters,
        "Estimated",
        "Fixed"
      ),
      IIV_CV_percent = IIV_CV_PERCENT
    ) |>
    select(
      Molecule,
      Parameter,
      Estimate,
      RSE_percent,
      Run4_status,
      IIV_CV_percent
    )
}

run4_parameter_inputs <- map_dfr(MOLECULES, extract_run4_parameters)

stopifnot(
  setequal(run4_parameter_inputs$Molecule, MOLECULES),
  all(is.finite(run4_parameter_inputs$Estimate)),
  all(run4_parameter_inputs$Estimate > 0),
  all(run4_parameter_inputs$IIV_CV_percent == 30)
)

run4_parameter_inputs |>
